```python
# app/api/endpoints/news.py
@router.get("/{id}")
async def read_news_detail(id: int, db: AsyncSession = Depends(get_db)):
    news = await crud_news.get_news_detail(db, news_id=id)
    if not news:
        raise HTTPException(status_code=404, detail={"code": 40404, "message": "资讯不存在"})
    
    # 添加阅读量+1逻辑
    news.view_count += 1
    await db.commit()
    await db.refresh(news)
    
    return {
        "code": 200,
        "message": "success",
        "data": news
    }
```

```python
# app/api/endpoints/users.py
def mask_phone(phone: str) -> str:
    if phone and len(phone) >= 7:
        return phone[:3] + '****' + phone[-4:]
    return phone

@router.get("/me")
async def get_current_user_info(current_user: User = Depends(get_current_user)):
    return {
        "code": 200,
        "message": "success",
        "data": {
            "id": current_user.id,
            "username": current_user.username,
            "phone": mask_phone(getattr(current_user, 'phone', None)),
            "elder_mode": getattr(current_user, 'elder_mode', False),
            "role": getattr(current_user, 'role', 'user'),
            "create_time": current_user.create_time.isoformat() + "Z" if current_user.create_time else None
        }
    }
```

#### 3. 修复管理员用户列表返回格式

```python
# app/api/endpoints/users.py
@router.get("/admin/users")
async def get_admin_users(
    skip: int = 0,
    limit: int = 20,
    keyword: str = None,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    if current_user.role != "admin":
        raise HTTPException(status_code=403, detail={"code": 40301, "message": "角色不匹配"})
    
    users, total = await crud_user.get_users(db, skip=skip, limit=limit, keyword=keyword)
    return {
        "code": 200,
        "message": "success",
        "data": {
            "total": total,
            "list": [
                {
                    "id": user.id,
                    "username": user.username,
                    "phone": mask_phone(user.phone),
                    "elder_mode": user.elder_mode,
                    "create_time": user.create_time.isoformat() + "Z" if user.create_time else None
                }
                for user in users
            ]
        }
    }
```
